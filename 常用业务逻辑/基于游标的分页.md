#### 简介

目前主流的分页模式有两种

- 基于偏移量的分页
- 基于游标的分页

#### 基于偏移量的分页

```
http://127.0.0.1:8080/forum/village/list?page=1
或
http://127.0.0.1:8080/forum/village/list?page=1&size=10
```

客户端需要提供本次请求每页所需的结果数（limit）和偏移量（offset），偏移量通常由服务端通过page和size计算得出

这种分页方式十分简单，只需跳过前面Offset指定的结果数，按需返回Limit个结果数就可以了

例如
```
第一页
SELECT * from hs_form_village ORDER BY id DESC LIMIT 10 OFFSET 0;

第二页
SELECT * from hs_form_village ORDER BY id DESC LIMIT 10 OFFSET 10;

第三页
SELECT * from hs_form_village ORDER BY id DESC LIMIT 10 OFFSET 10;
```

- 优势

1、简单

2、支持跳页访问


- 劣势

1、在数据量很大的场景下，查询效率会比较低

2、在并发场景下会出现元素重复，特别是数据新增或删除的操作频繁的情况

3、显式的page参数在支持跳页的同时也会被爬虫并发请求

- 其他补充

当分页变慢之后需要做一些优化，例如：

1、使用子查询
```
select * from hs_form_village 
where id in (
 select id from (
    select id from hs_form_village 
    where comment_num > 10 
    limit 100000,10
 ) as t
);

```

2、使用inner join关联查询

```
select * from hs_form_village 
inner join (
     select id from hs_form_village 
    where comment_num > 10 
    limit 100000,10
) as t on t.id=hs_form_village.id
```

3、使用以下介绍的分页游标方式

#### 基于游标的分页

基于游标的分页是指接口在返回响应数据的同时返回一个 cursor ，通常是一个不透明字符串。它表示的是这一页数据的最后那个元素，就好比书签存档，通过这个 cursor 接口就能准确的返回下一页的数据

例如
```
第一页
SELECT * from hs_form_village ORDER BY id DESC LIMIT 10;

第二页
SELECT * from hs_form_village where id < 10 ORDER BY id DESC LIMIT 10;

第三页
SELECT * from hs_form_village where id < 20 ORDER BY id DESC LIMIT 10;


或者多条件多排序：
第二页（记住第一页最后一条的price=143.0, id=7）
SELECT * FROM hs_marketing_goods
WHERE (status = 101 OR category_id = 8)
  AND (price < 143.0 OR (price = 143.0 AND id < 7))
ORDER BY price DESC, id DESC
LIMIT 10;

SELECT * FROM hs_marketing_goods
WHERE (status = 101 OR category_id in (1,2,8))
  AND (
    price < :last_price
    OR (price = :last_price AND category_id < :last_category_id)
    OR (price = :last_price AND category_id = :last_category_id AND id < :last_id)
  )
ORDER BY price DESC, category_id DESC, id DESC
LIMIT 10; 
```

- 优势

1、性能好

2、相对安全，防止被恶意爬虫


- 劣势

1、实现较复杂

2、不支持跳页访问

#### 基于游标的分页实现方案

基于游标的分页必须要指定排序字段，且查询的字段中必须包含排序的字段

现在我们实现一个 APP 滚动到底部后分页获取数据的例子，通常我们只需要获取下一页的数据，而不是上一页的数据，之后进行数据追加

- 分析

1、需要知道请求的时间，这个 TOKEN 生成的时间，不是一直有效，通常这个 TOKEN 是通过 JSON 序列化并使用 Base-64 编码的字符串

2、需要知道排序的字段是什么，排序规则是 DESC 还是 ASC

3、另外通常有 JOIN 操作，所以需要知道排序字段的表名是什么

4、记录上一页的索引值，是 > 还是 < ，另外主键是不可以等于的

```
type Paginator struct {
	Time       uint   //创建分页器的时间戳
	PrimaryKey string //记录行的主键

	Params []PaginatorParam //排序字段
}

// 排序字段信息
type PaginatorParam struct {
	Field     string `json:"field"`      //字段名
	Value     any    `json:"value"`      //值
	ValueType string `json:"value_type"` //值类型
	Direction string `json:"direction"`  //排序规则
	TableName string `json:"table"`      //表名
}
```

1、如上我们已经知道查询提交、分页数，以及排序规则，接下来需要把排序字段记录下来，然后下一页检查到 cursor 不为空时，说明接口需要返回第二页分页的数据

2、第二页追加查询条件

3、当获取到一页数据后，递归获取最后的数据作为书签记录下来

4、返回一个加密的 TOKEN 给客户端，获取下一页时需要提供

#### 完整代码

```
package util

import (
	"encoding/base64"
	"errors"
	"fmt"
	"strings"

	"cms/util/text"
	"cms/util/timex"

	"github.com/bytedance/sonic"
)

const (
	//最大有效间隔时间
	PAGINATOR_MAX_INTERVAL_TIME uint = 86400
)

// 获取向下滚动的分页器
// 1、需要有排序，注意排列顺序
// 2、查询的字段必须包含排序的字段，保证能正常填充数据
type Paginator struct {
	Time       uint   //创建分页器的时间戳
	PrimaryKey string //记录行的主键

	Params []PaginatorParam //排序字段
}

// 排序字段信息
type PaginatorParam struct {
	Field     string `json:"field"`      //字段名
	Value     any    `json:"value"`      //值
	ValueType string `json:"value_type"` //值类型
	Direction string `json:"direction"`  //排序规则
	TableName string `json:"table"`      //表名
}

// 创建实例
//
// 原理：
//
//	1、将排序字段参数记录到Paginator实例中
//	2、将结果集中的值对Pagerinator实例的字段进行赋值
//	3、生成游标加密字符串
//	如果有token就进行where条件追加，进行分页查询
func NewPaginator(timeFormatter *timex.TimeFormatter, where map[string]any, order []string, token string, primaryKeys ...string) (*Paginator, error) {
	page := &Paginator{}

	//主键
	if len(primaryKeys) == 0 {
		page.PrimaryKey = "id"
	}

	//解码游标加密字符串，并检查参数合法性
	err := page.decodeCursorToken(token, timeFormatter)
	if err != nil {
		return page, err
	}

	//当前时间
	page.Time = timeFormatter.UnixTime()

	if token == "" {
		//解析排序字段
		page.parsePaginatorOrders(order)
	} else {
		//追加查询条件
		page.appendQuery(where)
	}

	return page, nil
}

// 解析多个排序字段
func (p *Paginator) parsePaginatorOrders(orders []string) {
	for _, o := range orders {
		p.parsePaginatorOrder(o)
	}
}

// 解析排序字段
func (p *Paginator) parsePaginatorOrder(order string) {
	//逗号分隔多个条件
	orders := strings.Split(order, ",")
	for _, o := range orders {
		param := PaginatorParam{}

		//排序规则
		fieldOrder := strings.Split(o, " ")
		param.Direction = "asc"

		var fieldTable string
		if len(fieldOrder) == 2 {
			fieldTable = fieldOrder[0]
			param.Direction = strings.ToLower(fieldOrder[1])
		} else {
			fieldTable = o
		}

		//是否有表名
		ft := strings.Split(fieldTable, ".")
		if len(ft) == 2 {
			param.Field = ft[1]
			param.TableName = ft[0]
		} else {
			param.Field = fieldTable
		}

		//追加
		p.Params = append(p.Params, param)
	}
}

// appendQuery 追加分页查询条件
func (p *Paginator) appendQuery(where map[string]any) {
	if len(p.Params) == 0 {
		return
	}

	// 准备条件组
	var conditions []string
	var values []any

	// 生成每个OR分支的条件
	for i := 0; i < len(p.Params); i++ {
		var parts []string

		// 前面所有字段必须等于
		for j := 0; j < i; j++ {
			field := p.getFieldName(p.Params[j].Field, p.Params[j].TableName)
			parts = append(parts, fmt.Sprintf("%s = ?", field))
			values = append(values, p.Params[j].Value)
		}

		// 当前字段使用不等条件
		field := p.getFieldName(p.Params[i].Field, p.Params[i].TableName)
		op := p.getComparisonOperator(p.Params[i].Direction)
		parts = append(parts, fmt.Sprintf("%s %s ?", field, op))
		values = append(values, p.Params[i].Value)

		// 组合成AND条件
		condition := "(" + strings.Join(parts, " AND ") + ")"
		conditions = append(conditions, condition)
	}

	// 组合所有OR条件
	finalCondition := strings.Join(conditions, " OR ")

	// 添加到where map
	where[finalCondition] = values
}

// 获取比较操作符
func (p *Paginator) getComparisonOperator(direction string) string {
	if direction == "desc" {
		return "<"
	}
	return ">"
}

// 获取带表名的字段名
func (p *Paginator) getFieldName(field, tableName string) string {
	if tableName != "" {
		return fmt.Sprintf("%s.%s", tableName, field)
	}
	return field
}

// 设置游标对应的值
func (p *Paginator) SetCursorValue(data map[string]any) {
	for key, value := range data {
		for index, param := range p.Params {
			if param.Field == key {
				p.Params[index].Value = value
				p.Params[index].ValueType = text.GetComparableValueType(value)
			}
		}
	}
}

// 获取游标加密字符串
func (p *Paginator) GetCursorToken() string {
	data, err := sonic.Marshal(p)
	if err != nil {
		return ""
	}

	return base64.StdEncoding.EncodeToString([]byte(data))
}

// 解码游标加密字符串
// 如果有token就需要解析
func (p *Paginator) decodeCursorToken(token string, timeFormatter *timex.TimeFormatter) (err error) {
	if len(token) == 0 {
		return
	}

	//解码
	bytes, err := base64.StdEncoding.DecodeString(token)
	if err != nil {
		err = errors.New("请求查询参数异常")
		return
	}

	err = sonic.UnmarshalString(string(bytes), p)
	if err != nil {
		err = errors.New("请求查询参数异常")
		return
	}

	//时间是否过期
	now := timeFormatter.UnixTime()
	if p.Time > now || now-p.Time > PAGINATOR_MAX_INTERVAL_TIME {
		err = errors.New("请求查询参数异常")
		return
	}

	return
}
```
